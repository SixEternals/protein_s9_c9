# models/bl0_cclmoff.py
import sys
sys.path.insert(0, "/data/zwf/code1/reborn_seed")

import fm
import torch
import torch.nn as nn

class BL0_CCLMoff(nn.Module):
    """
    CCLMoff baseline: RNA-FM (12L/640d) + MLP
    Reference: https://github.com/duwa2/CCLMoff
    """
    def __init__(self, dropout=0.2):
        super().__init__()
        
        # RNA-FM: 12 layers, 640 dim, ~99M params
        self.rna_model, self.rna_alphabet = fm.pretrained.rna_fm_t12()
        
        # MLP head (exactly as CCLMoff)
        self.dense1 = nn.Linear(640, 64)
        self.dense2 = nn.Linear(64, 1)
        self.dropout = nn.Dropout(dropout)
        self.act = nn.Sigmoid()
        self.elu = nn.ELU()
        
        # Kaiming init (CCLMoff uses this)
        for m in self.children():
            if isinstance(m, nn.Linear):
                nn.init.kaiming_uniform_(m.weight.data)
    
    def get_alphabet(self):
        return self.rna_alphabet
    
    def forward(self, tokens):
        # tokens: [batch, seq_len] integer token IDs
        rna_results = self.rna_model(tokens, repr_layers=[12])
        seq_emb = rna_results["representations"][12]  # [batch, seq_len, 640]
        cls_emb = seq_emb[:, 0, :]  # CLS token [batch, 640]
        
        x = self.elu(self.dropout(self.dense1(cls_emb)))
        x = self.dense2(self.dropout(x))
        out = self.act(x)
        return out
